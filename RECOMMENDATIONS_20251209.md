def run_scraper_with_retry(scraper_func, max_retries=2):
    for attempt in range(max_retries):
        try:
            scraper_func()
            return True
        except Exception as e:
            if attempt < max_retries - 1:
                logger.warning(f"Attempt {attempt+1} failed, retrying...")
            else:
                logger.error(f"All attempts failed: {e}")
                return False

def deduplicate(products):
    unique = []
    seen = set()
    for p in products:
        key = (p['name'].lower().strip(), p['price'])
        if key not in seen:
            seen.add(key)
            unique.append(p)
    return unique
